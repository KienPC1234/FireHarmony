# Serving-only dependencies, for the container that runs the API + dashboard.
#
# Deliberately NOT requirements.txt: that file pins a CUDA 13 torch build and pulls
# trl, bitsandbytes, jupyter, opencv and the rest of the training stack, which would
# make the image several GB larger for nothing. This image serves; it does not train.
#
# The CPU wheel index comes first so pip prefers "+cpu" local versions over the
# default CUDA wheels on Linux.
--extra-index-url https://download.pytorch.org/whl/cpu

numpy<2.0
scikit-learn
joblib

sentence-transformers
transformers
faiss-cpu

fastapi
uvicorn[standard]
python-dotenv
pyyaml

# PostgreSQL driver for query audit / feedback storage.
psycopg[binary]>=3.2
# Connection pool, so an audited question reuses a connection instead of
# opening a new TCP connection to the server every time.
psycopg-pool>=3.2
